// Verification task 43: seeded-random sequences against the balance cache
// (LED-2, LED-3) with the real store, ledger service, collector and holds
// service. Not product code.
package collector

import (
	"errors"
	"fmt"
	"math/rand/v2"
	"os"
	"strconv"
	"strings"
	"testing"
	"time"

	"github.com/panset/hourglass/internal/holds"

type v43stats struct {
	ops, hard, window, windowOdd, gateChecks, gateFail, restarts, failedPolls, rollovers int
	opCount                                                                          map[string]int
}

var v43ops = []string{"poll", "grant", "neggrant", "reset", "roleoff", "roleon", "minutes", "days", "months", "restart", "failpoll"}

func v43periods(from, to string) []string {
	var out []string
	for p := from; p <= to; p = ledger.NextPeriod(p) {
		out = append(out, p)
	}
	return out
}

// v43run runs one seeded sequence; it returns hard failures (strings) and stats.
func v43run(t *testing.T, seed uint64, nops int, st *v43stats) []string {
	r := rand.New(rand.NewPCG(seed, 0x43))
	e := newEnv(t, false)
	start := time.Date(2026, time.Month(1+r.IntN(12)), 1+r.IntN(28), r.IntN(24), r.IntN(60), 0, 0, time.UTC)
	e.clk.Set(start)
	startPeriod := e.led.CurrentPeriod()
	e.setting(settings.KeyEnforcement, "true")
	e.setting(settings.KeyMaxHoldsPerUser, "5")
	e.rules("tier=gold", 2)
	admin := e.adminID()
	const nu = 5
	var users, setups []int64
	for i := 0; i < nu; i++ {
		users = append(users, e.user(fmt.Sprintf("u%d", i)))
		setups = append(setups, e.setup(fmt.Sprintf("lab-%02d", i), true, map[string]string{"tier": "gold"}))
		e.allowance(users[i])
		if r.IntN(3) > 0 {
			e.hold(setups[i], users[i])
		}
	}
	hasRole := map[int64]bool{}
	for _, u := range users {
		hasRole[u] = true
	}
	var fails []string
	var log []string
	fail := func(f string, a ...any) {
		msg := fmt.Sprintf("seed %d: ", seed) + fmt.Sprintf(f, a...) + "\n  ops: " + strings.Join(log, ", ")
		fails = append(fails, msg)
	}
	must := func(err error, what string) {
		if err != nil {
			fail("%s: %v", what, err)
		}
	}
	pending := func() bool {
		last, err := e.st.LatestAllowancePeriod(bg)
		must(err, "latest")
		return last != "" && last < e.led.CurrentPeriod()
	}
	snapshot := func() map[string]int64 {
		m := map[string]int64{}
		for _, u := range users {
			for _, p := range v43periods(startPeriod, e.led.CurrentPeriod()) {
				s, err := e.st.SumLedger(bg, u, p, "")
				must(err, "sum")
				m[fmt.Sprintf("%d/%s", u, p)] = s
			}
		}
		return m
	}
	check := func(step string) {
		cur := e.led.CurrentPeriod()
		pend := pending()
		last, _ := e.st.LatestAllowancePeriod(bg)
		for _, u := range users {
			x, err := e.st.GetUser(bg, u)
			must(err, "get user")
			sum, err := e.st.SumLedger(bg, u, cur, "")
			must(err, "sum")
			if x.BalanceMT == sum {
				continue
			}
			if !pend {
				st.hard++
				fail("after op %d %q at %s (period %s, role %v): user %d cache %d != ledger sum %d (diff %+d)",
					len(log), step, e.clk.Now().Format(time.RFC3339), cur, hasRole[u], u, x.BalanceMT, sum, x.BalanceMT-sum)
				continue
			}
			// Documented pending-rollover window (D-A, finding 7): a user with no
			// entry yet in the new period may still show the period the cache was for.
			st.window++
			n, _ := e.st.CountLedger(bg, u, cur, "")
			// the cache may hold any period between the last opened one and the
			// current one (the period that was current at the user's last append)
			explained := false
			var prev int64
			for _, p := range v43periods(last, cur) {
				if p == cur {
					break
				}
				ps, _ := e.st.SumLedger(bg, u, p, "")
				if p == last {
					prev = ps
				}
				if ps == x.BalanceMT {
					explained = true
				}
			}
			if n > 0 || !explained {
				st.windowOdd++
				fail("pending window, unexplained: after %q at %s: user %d cache %d, cur %s sum %d, entries in cur %d, last-opened %s sum %d",
					step, e.clk.Now().Format(time.RFC3339), u, x.BalanceMT, cur, sum, n, last, prev)
			}
		}
	}
	roleUsers := func(want bool) []int64 {
		var out []int64
		for _, u := range users {
			if hasRole[u] == want {
				out = append(out, u)
			}
		}
		return out
	}
	pick := func(xs []int64) (int64, bool) {
		if len(xs) == 0 {
			return 0, false
		}
		return xs[r.IntN(len(xs))], true
	}
	check("setup")
	for i := 0; i < nops; i++ {
		op := v43ops[r.IntN(len(v43ops))]
		st.ops++
		st.opCount[op]++
		desc := op
		switch op {
		case "poll":
			e.pressure.SetValue([]float64{0.6, 0.8, 0.95}[r.IntN(3)])
			// make sure someone holds something
			for j, u := range users {
				if hasRole[u] && r.IntN(2) == 0 {
					if _, err := e.st.ActiveHoldForSetup(bg, setups[j]); errors.Is(err, store.ErrNotFound) {
						e.hold(setups[j], u)
					}
				}
			}
			e.clk.Advance(poll)
			before, _ := e.st.LatestAllowancePeriod(bg)
			must(e.c.RunOnce(bg), "poll")
			after, _ := e.st.LatestAllowancePeriod(bg)
			if after != before {
				st.rollovers++
			}
		case "grant", "neggrant":
			u, ok := pick(roleUsers(true))
			if !ok {
				desc += "(skip)"
				break
			}
			amt := int64(1 + r.IntN(400000))
			if op == "neggrant" {
				amt = -amt
			}
			desc = fmt.Sprintf("%s u%d %d", op, u, amt)
			_, err := e.led.Grant(bg, admin, u, amt, "v43")
			must(err, "grant")
		case "reset":
			u, ok := pick(roleUsers(true))
			if !ok {
				desc += "(skip)"
				break
			}
			desc = fmt.Sprintf("reset u%d", u)
			_, err := e.led.Reset(bg, admin, u, "v43")
			must(err, "reset")
		case "roleoff":
			u, ok := pick(roleUsers(true))
			if !ok {
				desc += "(skip)"
				break
			}
			desc = fmt.Sprintf("roleoff u%d", u)
			hs, _ := e.st.UserHolds(bg, u)
			for _, h := range hs {
				if h.ReleasedAt == nil {
					must(e.st.ReleaseHold(bg, h.ID, &admin, "v43", e.clk.Now()), "release")
				}
			}
			must(e.st.Tx(bg, func(tx *store.Tx) error { return tx.SetUserRoles(bg, u, nil) }), "roleoff")
			hasRole[u] = false
		case "roleon":
			u, ok := pick(roleUsers(false))
			if !ok {
				desc += "(skip)"
				break
			}
			desc = fmt.Sprintf("roleon u%d", u)
			must(e.st.Tx(bg, func(tx *store.Tx) error {
				if err := tx.SetUserRoles(bg, u, []string{store.RoleDeveloper}); err != nil {
					return err
				}
				return e.led.EnsureFirstAllowanceTx(tx, u, e.clk.Now())
			}), "roleon")
			hasRole[u] = true
		case "minutes":
			d := time.Duration(1+r.IntN(120)) * time.Minute
			desc = fmt.Sprintf("+%s", d)
			e.clk.Advance(d)
		case "days":
			d := 1 + r.IntN(10)
			desc = fmt.Sprintf("+%dd", d)
			e.clk.Advance(time.Duration(d) * 24 * time.Hour)
		case "months":
			m := 1 + r.IntN(3)
			desc = fmt.Sprintf("+%dmo", m)
			e.clk.Set(e.clk.Now().AddDate(0, m, 0))
		case "restart":
			st.restarts++
			before := snapshot()
			e.led = &failingLedger{Service: ledger.NewService(e.st, e.rec, e.clk, time.UTC)}
			e.c = e.newCollector()
			if os.Getenv("V43_NOREPAIR") == "" {
				_, err := e.led.RepairBalances(bg) // app.Run does this before the collector and HTTP
				must(err, "repair")
			}
			after := snapshot()
			for k, v := range before {
				if after[k] != v {
					fail("restart changed ledger sum %s: %d -> %d", k, v, after[k])
				}
			}
		case "failpoll":
			st.failedPolls++
			e.clk.Advance(poll)
			e.led.setFail(errors.New("v43 injected"))
			if err := e.c.RunOnce(bg); err == nil {
				// no active hold: nothing to debit, the poll succeeds
				desc += "(no debit, ok)"
			}
			e.led.setFail(nil)
		}
		log = append(log, desc)
		check(desc)
		if len(fails) > 0 {
			return fails
		}
	}
	// Check 3: Acquire gating through the real holds service, enforcement on.
	hsvc := holds.NewService(holds.Deps{Store: e.st, Clock: e.clk})
	pend := pending()
	cur := e.led.CurrentPeriod()
	for j, u := range roleUsers(true) {
		sum, _ := e.st.SumLedger(bg, u, cur, "")
		x, _ := e.st.GetUser(bg, u)
		sid := e.setup(fmt.Sprintf("gate-%d-%d", seed, j), true, nil)
		h, err := hsvc.Acquire(bg, u, sid)
		var he *holds.Error
		got := err == nil
		if err != nil && !(errors.As(err, &he) && he.Code == holds.CodeInsufficientTokens) {
			fail("acquire u%d: unexpected error %v", u, err)
			continue
		}
		st.gateChecks++
		want := sum > 0
		if pend {
			want = x.BalanceMT > 0 // documented window: the cache still holds the opened period
		}
		if got != want {
			st.gateFail++
			fail("gating u%d: ledger sum %d, cache %d, pending %v: acquire allowed=%v", u, sum, x.BalanceMT, pend, got)
		}
		if got {
			must(e.st.ReleaseHold(bg, h.ID, &admin, "v43", e.clk.Now()), "release gate hold")
		}
	}
	return fails
}

func TestV43_RandomisedCacheEqualsLedger(t *testing.T) {
	n := 200
	if s := os.Getenv("V43_SEEDS"); s != "" {
		n, _ = strconv.Atoi(s)
	}
	base := uint64(43000)
	if s := os.Getenv("V43_BASE"); s != "" {
		b, _ := strconv.Atoi(s)
		base = uint64(b)
	}
	st := &v43stats{opCount: map[string]int{}}
	failed := 0
	for i := 0; i < n; i++ {
		seed := base + uint64(i)
		var fails []string
		t.Run(fmt.Sprintf("seed=%d", seed), func(t *testing.T) {
			fails = v43run(t, seed, 60, st)
			for _, f := range fails {
				t.Error(f)
			}
		})
		if len(fails) > 0 {
			failed++
		}
	}
	t.Logf("seeds %d failed %d; ops %d %v; hard %d; window mismatches %d (unexplained %d); restarts %d; failed polls %d; rollover polls %d; gating checks %d failed %d",
		n, failed, st.ops, st.opCount, st.hard, st.window, st.windowOdd, st.restarts, st.failedPolls, st.rollovers, st.gateChecks, st.gateFail)
}
