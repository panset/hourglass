import os
only = sys.argv[2].split(',') if len(sys.argv) > 2 and sys.argv[2] else None
extra_pkgs = sys.argv[3].split(',') if len(sys.argv) > 3 else []
for name, f, old, new, pkgs in M:
    if only and name.split()[0] not in only: continue
    pkgs = pkgs + extra_pkgs
    name = name + (' [+' + ','.join(extra_pkgs) + ']' if extra_pkgs else '')
    p = root / f
    src = p.read_text()
    if src.count(old) != 1:
        print(f"{name}: PATTERN NOT FOUND ({src.count(old)})"); continue
    extra = ""
    if "boolI" in new:
        extra = "\nfunc boolI(b bool) int64 { if b { return 1 }; return 0 }\n"
    p.write_text(src.replace(old, new) + extra)
    try:
        r = subprocess.run(["go","test","-count=1"] + [f"./internal/{x}/" for x in pkgs], cwd=root, capture_output=True, text=True, timeout=600)
        out = r.stdout + r.stderr
        fails = [l.strip() for l in out.splitlines() if l.startswith("--- FAIL") or "build failed" in l or "[build failed]" in l]
        verdict = "KILLED" if r.returncode != 0 else "SURVIVED"
        print(f"{name}: {verdict}; {fails[:4]}")
    finally:
        p.write_text(src)
